// NBC: the Super Shorty pack's DECORATE, kept as an unused reference file (not included). The weapon is NBC_SuperShorty in zscript/nbc/weapons/firearms.zs.
ACTOR SuperShorty : Weapon
{
    +FLOORCLIP
	+AMMO_CHECKBOTH
	+NOAUTOFIRE
	+NoAlert
	Weapon.SlotNumber 3
    //Weapon.AmmoType "Shell" //Added these in case someone wants to disable reloading altogether.
    //Weapon.AmmoGive 3 //Backdash the three AmmoType sections and the +AMMO_CHECKBOTH flags below.
    Weapon.AmmoType1 "ShortyLoaded"
    Weapon.AmmoType2 "Shell"
    Weapon.AmmoGive2 5
    Weapon.AmmoUse1 1
    Weapon.AmmoUse2 1
    Weapon.KickBack 400
	Tag "Super Shorty"
    Inventory.PickupSound "Misc/w_pkup" 
    Inventory.PickupMessage "Picked up a Super Shorty."
    States
    {
    Spawn:
        SERC A -1
        Loop
    Ready:
        SERB A 1 A_WeaponReady(WRF_ALLOWRELOAD) //Remove this flag to remove manual reloading.
        Loop
    Deselect:
        SERB A 1 A_Lower
        NULL AA 0 A_Lower
        Loop
    Select:
        SERB A 1 A_Raise
        NULL AA 0 A_Raise
        Loop
    Fire:
        NULL A 0 A_JumpIf(CountInv("ShortyLoaded") == 0,"Reload") //Backdash this to disable the reload check.
        NULL A 0 A_JumpIf(CountInv("ShortyLoading") == 1,"ReloadFinish2")
		NULL A 0 A_GunFlash
		NULL A 0 A_AlertMonsters
        NULL A 0 A_PlayWeaponSound("SERBFIR")
        SERF A 1 BRIGHT A_FireBullets(8.5, 8.5, 9, 8, "ShortyPuff", 1)
	    //NULL A 0 A_ZoomFactor(0.975) //Remove the backdashes to enable visual recoil.
        //NULL A 0 A_SetPitch(Pitch -1.8)
	    //NULL A 0 A_Quake(8, 6, 0, 48576, "None")
        SERF B 1 BRIGHT
		SERF CD 2
	    //NULL A 0 A_ZoomFactor(1.000)
		NULL A 0 A_JumpIf(CountInv("ShortyLoaded") == 0,"EmptyOut") //Backdash this to skip the empty chamber token altogether.
		SERB A 8
		Goto Pump
		Pump:
		SERB BC 1
		NULL A 0 A_PlaySoundEX("SERBPM1",CHAN_6)
        SERB D 1
		SERB FE 2
		NULL A 0 A_PlaySoundEX("SERBPM2",CHAN_7)
		SERB DCB 2
		SERB A 0 A_Refire
        goto Ready
		EmptyOut:
		SERB A 4 A_GiveInventory("ShortyEmpty",1)
		Goto Reload
    Reload:
        NULL A 0 A_JumpIfInventory("ShortyLoaded", 3, 2)
        NULL A 0 A_JumpIfInventory("Shell", 1, "ReloadWork")
        SERB A 1 A_WeaponReady
        Goto Ready
	ReloadWork:
	    SERB A 0 A_GiveInventory("ShortyLoading",1)
        SERB ABCD 1
		SERR AB 1
  ReloadLoop:
 		NULL A 0 A_PlaySoundEX("SERBREL",CHAN_WEAPON)
		SERR CDE 2
        NULL A 0 A_TakeInventory("Shell", 1,  TIF_NOTAKEINFINITE)
        NULL A 0 A_GiveInventory("ShortyLoaded", 1)
        NULL A 0 A_JumpIfInventory("ShortyEmpty", 1, "DryPump") //Backdash this to disable the chambering animation.
        SERR B 4 A_WeaponReady(WRF_NOBOB|WRF_NOSWITCH)
        NULL A 0 A_JumpIfInventory("ShortyLoaded", 3, "ReloadFinish")
        NULL A 0 A_JumpIfInventory("Shell", 1, "ReloadLoop")
        Goto ReloadFinish
  ReloadFinish:
	    NULL A 0 A_TakeInventory("ShortyLoading",99)
        SERR A 1
        SERB DCBA 2
		NULL A 0 A_Refire("Fire")
        Goto Ready
  ReloadFinish2:
	    NULL A 0 A_TakeInventory("ShortyLoading",99)
        SERR A 1
        SERB DCB 1
		NULL A 0 A_Refire("Fire")
        Goto Ready
	DryPump:
	    SERR B 3
		SERR BA 2
		SERB D 1
		NULL A 0 A_PlaySoundEX("SERBPM1",CHAN_6)
		SERB D 1 A_TakeInventory("ShortyEmpty",99)
		SERB F 2
		SERB E 1
		NULL A 0 A_PlaySoundEX("SERBPM2",CHAN_7)
		SERB E 1
		SERB D 3
		SERR AB 2
		SERR B 5
		Goto ReloadLoop
}
}

ACTOR ShortyLoaded : Ammo
{
   +IGNORESKILL
   Inventory.Amount 0
   Inventory.MaxAmount 3
   Ammo.BackpackAmount 0
   Ammo.BackpackMaxAmount 3
}

ACTOR ShortyEmpty : Inventory { Inventory.MaxAmount 1 }
ACTOR ShortyLoading : Inventory { Inventory.MaxAmount 1 }
