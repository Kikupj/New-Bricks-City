actor HitmanTommy : HumanMb
{
  scale 1.0
  obituary "%o was riddled by a Mob Hitman."
  health 70
  radius 20
  height 56
  mass 100
  speed 8
  painchance 200
  seesound "mobs/See"
  attacksound "en/ak47"
  painsound "mobs/pain"
  deathsound "mobs/die"
  activesound ""
  dropitem "Chaingun" 256
  Tag "Mob Hitman"
  Translation "112:127=64:79"
  DamageFactor "Enemy", 0.0
  MONSTER
  +FLOORCLIP
  +NOTARGET
  +NOINFIGHTING
  states
  {
  Spawn:
    EC3A T 10 A_Look
    loop
  See:
    EC3A AABBCCDD 3 A_Chase
    loop
  Missile:
    EC3A EE 5 A_FaceTarget
	EC3A F 0 A_PlayWeaponSound("en/ak47")
    EC3A F 2 BRIGHT A_CustomMissile("EnemySMGTracer",32,0,random(-8,8))
    EC3A E 2 A_SentinelRefire
    goto Missile+2
  Pain:
    EC3A H 3
    EC3A H 3 A_Pain
    goto See
  Death:
   	TNT1 A 0 A_Jump(80, "ADeath1")
    EC2A I 5
    EC2A J 5 A_Scream
    EC2A K 5 A_NoBlocking
    EC2A L 5
    EC2A M -1
    stop
  ADeath1:
    EC2A N 5
    EC2A O 5 A_Scream
    EC2A P 5 A_NoBlocking
    EC2A Q 5
    EC2A R -1
    stop
  Death.Weak:
    EC2B I 6
    EC2B J 6 A_PlaySound("mobs/pain")
    EC2B K 6 A_NoBlocking
    EC2B L 6
    EC2B M -1
    stop
  Idle:
    EC3A AA 4 A_Wander
    EC3A A 0 A_Look
    EC3A BB 4 A_Wander
    EC3A A 0 A_Look
    EC3A CC 4 A_Wander
    EC3A A 0 A_Look
    EC3A DD 4 A_Wander
    EC3A A 0 A_Look
    loop
  }
}
