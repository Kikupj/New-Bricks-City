
actor Gnome1
{
  obituary "%o was blasted by a urban Gnome."
  health 60
  painchance 180
  speed 8
  radius 20
  height 56
  mass 100
  seesound "Gnome/see"
  painsound "gnomeP"
  deathsound "Gnome/die"
  Tag "Gnome"
  dropitem "Shotgun" 256
  Translation "192:207=96:111"
  DamageFactor "Enemy", 0.0
  MONSTER
  +FLOORCLIP
  +NOTARGET
  +NOINFIGHTING
  states
  {
  Spawn:
    EO2A T 10 A_Look
    Loop
  See:
	EO2A AABBCCDD 2 A_Chase
	loop
Missile:
    EO2A EF 10 A_FaceTarget
	EO2A G 0 A_PlayWeaponSound("en/shotgun")
    EO2A G 8 BRIGHT A_CustomMissile("EnemyLMGTracer",32,0,random(-1,1))
    EO2A F 1
	EO2A H 0 A_PlayWeaponSound("en/shotgun")
    EO2A H 8 BRIGHT A_CustomMissile("EnemyLMGTracer",32,0,random(-1,1))
	EO2A FE 4
    goto See
  Pain:
    EO2A I 2
    EO2A I 2 A_Pain
    goto See
  Death:
    EO2A J 8
    EO2A K 8 A_Scream
    EO2A L 6
	EO2A M 6
    EO2A N -1 A_NoBlocking
    stop
  Death.Weak:
    EO2B I 8
    EO2B J 8 A_Scream
    EO2B K 6
	EO2B L 6
    EO2B M -1 A_NoBlocking
    stop
	Death.Fire:
		EF1A A 5 BRIGHT
		EF1A B 5 BRIGHT A_PlaySound("Gnome/die")
		EF1A CDCB 5 BRIGHT 
		EF1A ABCD 5 BRIGHT
		EF1A E 5 BRIGHT A_NoBlocking
		EF1A EFGH 5 BRIGHT
		EF1A I -1
		Stop
  Idle:
    EO2A AA 4 A_Wander
    EO2A A 0 A_Look
    EO2A BB 4 A_Wander
    EO2A A 0 A_Look
    EO2A CC 4 A_Wander
    EO2A A 0 A_Look
    EO2A DD 4 A_Wander
    EO2A A 0 A_Look
    loop
  }
}

	DamageFactor "Enemy", 0.0
	Tag "Drone"
	MONSTER
	+FLOAT
	+NOGRAVITY
	+NOTARGET
	+NOINFIGHTING

	+FLOATBOB

	Spawn:

		EO1A H 4 A_Pain
		Goto See

	}
