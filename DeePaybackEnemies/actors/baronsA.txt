actor BadassElite
{
  scale 1.0
  obituary "%o was shredded by a Elite Badass"
  health 400
  radius 20
  height 56
  mass 200
  speed 10
  painchance 110
  seesound "badass/see"
  painsound "badass/pain"
  deathsound "badass/DIE"
  activesound " "
  dropitem "PlasmaRifle"
  Tag "Badass Elite"
  Translation "112:127=174:191"
  DamageFactor "Enemy", 0.0
  DamageFactor "Fire", 0.0
  DamageFactor "Weak", 0.5
  MONSTER
  +FLOORCLIP
  +NOTARGET
  +NOINFIGHTING
  +BOSS
  states
  {
  Spawn:
    EB1A S 10 A_Look
    loop
  See:
    EB1A AABBCCDD 3 A_Chase
    loop
  Missile:
    EB1A EE 5 A_FaceTarget
	EB1A F 0 A_PlayWeaponSound("en/m4car")
    EB1A F 2 BRIGHT A_CustomMissile("EnemySMGTracer",38,0,random(-4,4))
    EB1A E 4 A_SentinelRefire
    goto Missile+2
  Pain:
    EB1A H 3
    EB1A H 3 A_Pain
    goto See
  Death:
    EB1A I 5
    EB1A J 5 A_Scream
    EB1A K 5 A_BossDeath
    EB1A L 5 A_NoBlocking
	EB1A M -1
    stop
  Death.Weak:
    EB1B I 6
    EB1B J 6 A_PlaySound("badass/pain")
    EB1B K 6 A_BossDeath
    EB1B L 6 A_NoBlocking
	EB1B M -1
    stop
  Idle:
    EB1A AA 4 A_Wander
    EB1A A 0 A_Look
    EB1A BB 4 A_Wander
    EB1A A 0 A_Look
    EB1A CC 4 A_Wander
    EB1A A 0 A_Look
    EB1A DD 4 A_Wander
    EB1A A 0 A_Look
    loop
  }
}
